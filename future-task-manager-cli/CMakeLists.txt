cmake_minimum_required(VERSION 3.16)
project(future-task-manager-cli VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CURSES_NEED_NCURSES TRUE)
set(CURSES_NEED_WIDE TRUE)
find_package(Curses REQUIRED)
add_executable(future-task-manager-cli src/main.cpp)
target_include_directories(future-task-manager-cli PRIVATE ${CURSES_INCLUDE_DIRS})
target_link_libraries(future-task-manager-cli PRIVATE ${CURSES_LIBRARIES})
include(GNUInstallDirs)
install(TARGETS future-task-manager-cli DESTINATION ${CMAKE_INSTALL_BINDIR})
enable_testing()
add_test(NAME version_check COMMAND future-task-manager-cli --version)
